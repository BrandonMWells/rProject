//morocc_in,73,38,6	script	Aragham	4W_M_03,
//payon,144,173,5	script	Antonio	4_M_ORIENT01,
//alberta_in,28,58,0	script	Fredrik	4_M_03,
//yuno_in01,171,21,4	script	Lambert	4_M_ORIENT01,

prt_in,63,60,0	script	Blacksmith	63,{
	 disable_items;
	.@npc_name$ = "Blacksmith";
	.@price = 0;
	.@mineralcostmultiplier = 1;
	//.@canfail = 0;
	.@same = 0; // Check if the materials has the same refinement level
	.@startsafe = 1; // Start counting materials after the safe refine on Hollgrehenn (4 at Weapon Lvl 4 and Armor Lvl 1)
	

	mes "["+ .@npc_name$ +"]";
	mes "Hello, I can safely refine any of your equipment using expensive materials. Safe refining requires additional copies of the item you are refining.";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "I don't think I can refine any items you have...";
		close;
	}
	.@part = .@indices[select(.@menu$)];

	// custom check
	if(!getequipisequiped(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "You're not wearing";
		mes "anything there that";
		mes "I can refine.";
		emotion ET_FRET;
		close;
	}
	//Check if the item is refinable...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "I don't think I can";
		mes "refine this item at all...";
		close;
	}
	//Check to see if the items is already +10
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "I can't refine this";
		mes "any more. This is as";
		mes "refined as it gets!";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);



	.@count = 1;
	
 		switch (getequipweaponlv(.@part)) {

		case 1: 
			.@startref = 7; 
			.@w1[0] = 125;
			.@mcm1[0] = 1;
			setarray .@w1[0],250,500,1000,2000,3000,4000,5000,10000,15000,20000;
			setarray .@mcm1[0],1,1,1,1,1,1,1,1,2,3;
			.@price = .@w1[.@refinerycnt];
			.@mineralcostmultiplier = .@mcm1[.@refinerycnt];
			if (.@refinerycnt < .@startref) 
			.@count = 0;
			break;
						
		case 2: 
			.@startref = 6; 
			.@w2[0] = 250;
			.@mcm2[0] = 1;
			setarray .@w2[0],500,1000,2000,3000,4000,5000,10000,15000,20000,40000;
			setarray .@mcm2[0],1,1,1,1,1,1,1,2,3,4;
			.@price = .@w2[.@refinerycnt];
			.@mineralcostmultiplier = .@mcm2[.@refinerycnt];
			if (.@refinerycnt < .@startref) 
			.@count = 0;
			break;
		case 3: 
			.@startref = 5; 
			.@w3[0] = 125;
			.@mcm3[0] = 1;
			setarray .@w3[0],1000,2000,3000,4000,5000,10000,15000,20000,40000,80000;
			setarray .@mcm3[0],1,1,1,1,1,1,2,3,4,5;
			.@price = .@w3[.@refinerycnt];
			.@mineralcostmultiplier = .@mcm3[.@refinerycnt];
			if (.@refinerycnt < .@startref) 
			.@count = 0;
			break;
		case 4:
		default: 
		.@startref = 4; 
		.@w4[0] = 125;
		.@mcm4[0] = 1;
		setarray .@w4[0],1000,2000,4000,8000,10000,15000,20000,40000,80000,100000;
		setarray .@mcm4[0],1,1,1,1,1,2,3,4,5,6;
		.@price = .@w4[.@refinerycnt];
		.@mineralcostmultiplier = .@mcm4[.@refinerycnt];
		if (.@refinerycnt < .@startref) 
		.@count = 0;	
		break; // Armor
	} 
	
	
	mes "["+ .@npc_name$ +"]";
	mes "To refine this I need:";
	if(.@count > 0) {
		if (.@same == 1) mes "^4444aa> "+(.@count)+" pcs of +"+(.@refinerycnt)+" ^003366"+getitemname(.@refineitemid)+"^000000";
	else mes "^4444aa> "+(.@count)+" pcs of ^003366"+getitemname(.@refineitemid)+"^000000";
	}
	if (.@mineralcostmultiplier > 0) mes "^4444aa> "+(.@mineralcostmultiplier)+" pcs of "+getitemname(.@material)+"^000000";
	mes "^4444aa> A service fee of " + .@price + " Zeny.^000000";
	mes "Do you really wish to continue?";
	next;
	if(select("No:Yes") == 1){
		mes "["+ .@npc_name$ +"]";
		mes "Yeah...";
		mes "There's no need to";
		mes "rush. Take your time.";
		close;
	}

	// Checking materials
	set .@matcount,0;
	set .@hascard,0;
	getinventorylist;
	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
		if (.@same) {
			if (@inventorylist_id[.@i] == .@refineitemid && !@inventorylist_equip[.@i] && @inventorylist_refine[.@i] == .@refinerycnt) {
				.@matcount++;
				if (@inventorylist_card1[.@i] > 0) .@hascard = 1;
			}
		} else {
			if (@inventorylist_id[.@i] == .@refineitemid && !@inventorylist_equip[.@i]) {
				.@matcount++;
				if (@inventorylist_card1[.@i] > 0) .@hascard = 1;
			}
		}
		if (.@matcount == .@count) break;
	}
	if(.@matcount < .@count) {
		mes "["+ .@npc_name$ +"]";
		mes "You don't seem to fulfill any of the requirements needed to refine this equipment yet.";
		mes "Go get more materials with the same refinement levels.";
		close;
	}
	if(.@hascard == 1) {
		mes "["+ .@npc_name$ +"]";
		mes "^ff4444One of the materials (except the equipped one) contains a card on it!^000000";
		mes "^ff4444IMPORTANT NOTE: The cards on all the materials (except the equipped one) will be destroyed.^000000";
		mes "Do you really wish to continue?";
		next;
		
		if(select("No:Yes") == 1) {
			mes "["+ .@npc_name$ +"]";
			mes "Yeah...";
			mes "There's no need to";
			mes "rush. Take your time.";
			close;
		}
	}
	if(.@mineralcostmultiplier > 0 && countitem(.@material) < .@count*.@mineralcostmultiplier) {
		mes "["+ .@npc_name$ +"]";
		mes "You don't seem to fulfill any of the requirements needed to refine this equipment yet.";
		mes "Go get more "+getitemname(.@material)+" to proceed.";
		close;
	}
	if(Zeny < .@price) {
		mes "["+ .@npc_name$ +"]";
		mes "You don't seem to fulfill any of the requirements needed to refine this equipment yet.";
		mes "Go get more Zeny to proceed.";
		close;
	}
	
	// Payment
	set .@delcount,0;
	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
		if (.@same) {
			if (.@delcount < .@count && @inventorylist_id[.@i] == .@refineitemid && !@inventorylist_equip[.@i] && @inventorylist_refine[.@i] == .@refinerycnt) {
				delitemidx @inventorylist_idx[.@i], 1;
				set .@delcount,.@delcount+1;
			}
		} else {
			if (.@delcount < .@count && @inventorylist_id[.@i] == .@refineitemid && !@inventorylist_equip[.@i]) {
				delitemidx @inventorylist_idx[.@i], 1;
				set .@delcount,.@delcount+1;
			}
		}
	}
	if (.@mineralcostmultiplier > 0) {
		delitem .@material,.@mineralcostmultiplier;
		//delitem .@material,.@count*.@mineralcostmultiplier;
	}
	Zeny = Zeny-.@price;

	// Anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
	    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "["+ .@npc_name$ +"]";
		emotion ET_FRET;
		mes "Cheeky little shit";
		close;
	}


	// Success refining
	mes "["+ .@npc_name$ +"]";
	successrefitem .@part;
	emotion ET_SMILE;
	//specialeffect EF_NPC_EARTHQUAKE;
	close;
}
